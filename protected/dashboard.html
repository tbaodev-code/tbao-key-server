5. BẢNG MÃ PHẢN HỒI (HTTP STATUS CODES)

        

          

            
              

                MÃ HTTP
                TRẠNG THÁI
                Ý NGHĨA & HÀNH ĐỘNG
              

            
            
              

                200 OK
                valid: true
                Bản quyền hợp lệ! Kèm chữ ký số Ed25519 và HMAC-SHA256 để client xác minh tính toàn vẹn.
              

              

                400 BAD REQ
                REPLAY_ATTACK_DETECTED
                Phát hiện tấn công phát lại gói tin (Nonce đã từng được sử dụng). Yêu cầu bị hủy bỏ.
              

              

                400 BAD REQ
                TIMESTAMP_EXPIRED
                Thời gian của client lệch quá 5 phút so với đồng hồ Server. Kiểm tra lại giờ hệ thống.
              

              

                403 FORBIDDEN
                DEVICE_LIMIT_REACHED
                Máy lạ vượt quá giới hạn thiết bị. Máy chính chủ vẫn dùng bình thường, key không bị ban.
              

              

                403 FORBIDDEN
                IP_TEMPORARILY_BLOCKED
                Địa chỉ IP bị tạm khóa 15 phút do phát hiện hành vi Brute Force dò key trái phép liên tục.
              

              

                403 FORBIDDEN
                KEY_BANNED
                Key đã bị quản trị viên khóa thủ công. Thông báo cho người dùng.
              

              

                403 FORBIDDEN
                KEY_EXPIRED
                Key đã hết hạn sử dụng. Yêu cầu khách gia hạn.
              

              

                404 NOT FOUND
                KEY_NOT_FOUND
                Mã key không tồn tại trong hệ thống.

using System;
using System.Net.Http;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Threading.Tasks;
using Microsoft.Win32;

public class LicenseValidator
{
    private static readonly HttpClient client = new HttpClient();
    private const string API_URL = "http://localhost:3000/api/v1/check-key";

    // 1. TỰ ĐỘNG LẤY MÃ MÁY PHẦN CỨNG BẤT BIẾN (KHÔNG ĐỔI KHI CÀI LẠI WIN/ĐỔI MẠNG)
    public static string GetHardwareHWID()
    {
        string uuid = "", cpu = "", board = "";
        try
        {
            using (var key = Registry.LocalMachine.OpenSubKey(@"HARDWARE\DESCRIPTION\System\BIOS"))
                board = key?.GetValue("BaseBoardSerialNumber")?.ToString() ?? "";
            using (var key = Registry.LocalMachine.OpenSubKey(@"HARDWARE\DESCRIPTION\System\CentralProcessor\0"))
                cpu = key?.GetValue("ProcessorNameString")?.ToString() ?? "";
            using (var key = Registry.LocalMachine.OpenSubKey(@"SOFTWARE\Microsoft\Cryptography"))
                uuid = key?.GetValue("MachineGuid")?.ToString() ?? "";
        }
        catch { }

        string raw = $"{uuid}#{cpu}#{board}";
        using (var sha = SHA256.Create())
        {
            byte[] hash = sha.ComputeHash(Encoding.UTF8.GetBytes(raw));
            return "HWID-" + BitConverter.ToString(hash).Replace("-", "").Substring(0, 16);
        }
    }

    // 2. XÁC THỰC BẢN QUYỀN VỚI MÁY CHỦ TBAO KEY SERVER (CHỐNG REPLAY)
    public static async Task<bool> CheckLicense(string licenseKey)
    {
        string hwid = GetHardwareHWID();
        string nonce = Guid.NewGuid().ToString("N");
        long timestamp = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds();

        var payload = new { key = licenseKey, hwid = hwid, nonce = nonce, timestamp = timestamp };
        var content = new StringContent(JsonSerializer.Serialize(payload), Encoding.UTF8, "application/json");

        try
        {
            var response = await client.PostAsync(API_URL, content);
            string json = await response.Content.ReadAsStringAsync();
            using var doc = JsonDocument.Parse(json);
            var root = doc.RootElement;

            if (response.IsSuccessStatusCode && root.GetProperty("valid").GetBoolean())
            {
                // Kiểm tra Anti-Replay: Nonce trả về phải trùng khớp với Nonce vừa gửi đi
                if (root.TryGetProperty("nonce", out var n) && n.GetString() != nonce)
                {
                    Console.WriteLine("❌ CẢNH BÁO: Phát hiện tấn công Replay (Nonce không khớp)!");
                    return false;
                }

                Console.WriteLine("✅ Bản quyền hợp lệ!");
                if (root.TryGetProperty("expires_at", out var exp) && exp.ValueKind != JsonValueKind.Null)
                    Console.WriteLine("Hạn dùng: " + exp.GetString());
                else
                    Console.WriteLine("Bản quyền vĩnh viễn.");
                return true;
            }
            else
            {
                string reason = root.TryGetProperty("reason", out var r) ? r.GetString() : "UNKNOWN";
                Console.WriteLine("❌ Lỗi bản quyền: " + reason);
                return false;
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine("⚠️ Lỗi kết nối máy chủ: " + ex.Message);
            return false;
        }
    }
}

        


        
        

          
#include <windows.h>
#include <winhttp.h>
#include <iostream>
#include <string>

#pragma comment(lib, "winhttp.lib")

// Xác thực bản quyền bằng WinHTTP thuần (Không phụ thuộc thư viện ngoài)
bool CheckLicense(const std::wstring& serverHost, int port, const std::string& key, const std::string& hwid)
{
    HINTERNET hSession = WinHttpOpen(L"TBAO-GameClient/2.0", WINHTTP_ACCESS_TYPE_DEFAULT_PROXY, NULL, NULL, 0);
    if (!hSession) return false;

    HINTERNET hConnect = WinHttpConnect(hSession, serverHost.c_str(), port, 0);
    if (!hConnect) { WinHttpCloseHandle(hSession); return false; }

    HINTERNET hRequest = WinHttpOpenRequest(hConnect, L"POST", L"/api/v1/check-key",
                                            NULL, WINHTTP_NO_REFERER, WINHTTP_DEFAULT_ACCEPT_TYPES, 0);
    if (!hRequest) { WinHttpCloseHandle(hConnect); WinHttpCloseHandle(hSession); return false; }

    std::string jsonPayload = "{\"key\":\"" + key + "\",\"hwid\":\"" + hwid + "\"}";
    LPCWSTR headers = L"Content-Type: application/json\r\n";

    BOOL bSend = WinHttpSendRequest(hRequest, headers, -1L, (LPVOID)jsonPayload.c_str(), 
                                    (DWORD)jsonPayload.length(), (DWORD)jsonPayload.length(), 0);
    if (bSend) bSend = WinHttpReceiveResponse(hRequest, NULL);

    DWORD statusCode = 0;
    DWORD size = sizeof(statusCode);
    if (bSend) {
        WinHttpQueryHeaders(hRequest, WINHTTP_QUERY_STATUS_CODE | WINHTTP_QUERY_FLAG_NUMBER,
                            NULL, &statusCode, &size, NULL);
    }

    WinHttpCloseHandle(hRequest);
    WinHttpCloseHandle(hConnect);
    WinHttpCloseHandle(hSession);

    if (statusCode == 200) {
        std::cout << "[+] Kich hoat ban quyen thanh cong! (200 OK)\n";
        return true;
    } else {
        std::cout << "[-] Ban quyen tu choi! Ma loi HTTP: " << statusCode << "\n";
        return false;
    }
}

import subprocess
import hashlib
import requests

API_URL = "http://localhost:3000/api/v1/check-key"

# 1. LẤY MÃ MÁY PHẦN CỨNG BẤT BIẾN (BO MẠCH CHỦ + CPU ID + SERIAL)
def get_hardware_hwid() -> str:
    cmd = [
        "powershell", "-NoProfile", "-Command",
        "(Get-CimInstance Win32_ComputerSystemProduct).UUID; (Get-CimInstance Win32_Processor).ProcessorId; (Get-CimInstance Win32_BaseBoard).SerialNumber"
    ]
    try:
        out = subprocess.check_output(cmd, stderr=subprocess.DEVNULL).decode('utf-8', errors='ignore').strip().splitlines()
        raw = "#".join(out)
    except Exception:
        raw = "FALLBACK-HARDWARE"
    return "HWID-" + hashlib.sha256(raw.encode()).hexdigest()[:16].upper()

# 2. XÁC THỰC BẢN QUYỀN VỚI SERVER
def validate_key(key: str) -> bool:
    hwid = get_hardware_hwid()
    payload = {"key": key, "hwid": hwid}
    try:
        res = requests.post(API_URL, json=payload, timeout=10)
        data = res.json()
        
        if res.status_code == 200 and data.get("valid") is True:
            print("✅ Bản quyền hợp lệ!")
            print(f"Hạn dùng: {data.get('expires_at') or 'Vĩnh viễn ∞'}")
            print(f"Số máy đã kích hoạt: {data.get('bound_devices')}/{data.get('device_limit')}")
            return True
        else:
            print(f"❌ Từ chối: {data.get('reason')} - {data.get('message', '')}")
            return False
    except Exception as e:
        print(f"⚠️ Lỗi kết nối máy chủ: {e}")
        return False

# Thử nghiệm kiểm tra key
if __name__ == "__main__":
    validate_key("TBAO-XXXX-XXXX-XXXX")

      method: 'POST',
      headers: { 'Content-Type': 'application/json' },

    const data = await res.json();
    if (res.ok && data.valid) {
      console.log('✅ License hợp lệ!', data);
      return true;
    } else {
      console.error('❌ Từ chối:', data.reason, data.message);
      return false;
    }
  } catch (err) {
    console.error('⚠️ Không thể kết nối tới server bản quyền:', err);
    return false;
  }
}
